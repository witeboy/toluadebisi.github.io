function currentTimelineTime(){
  if (state.audioCtx && state.playbackNodes.length) return state.playbackStartTimelineTime + (state.audioCtx.currentTime-state.playbackStartCtxTime);
  return Number.isFinite(els.media.currentTime)?els.media.currentTime:0;
}

async function startPlayback(from=0){
  stopPlayback(false); const ctx=await ensureAudioContext(); from=clamp(from,0,projectEnd());
  state.playbackStartTimelineTime=from;state.playbackStartCtxTime=ctx.currentTime;
  scheduleRecordingTracks(ctx,from);
  if(state.sourceObjectUrl && from < (state.duration||Infinity)) { els.media.currentTime=clamp(from,0,Math.max(0,state.duration-.01)); await els.media.play().catch(()=>{}); }
  els.playBtn.textContent='❚❚'; animate();
}
function scheduleRecordingTracks(ctx,from){
  const anySolo=state.tracks.some(t=>t.solo);
  for(const track of state.tracks){
    if(track.muted || (anySolo&&!track.solo))continue;
    for(const clip of track.clips){
      const dur=clipDuration(clip), clipEnd=clip.start+dur; if(clipEnd<=from)continue;
      const localOffset=Math.max(0,from-clip.start); const when=Math.max(0,clip.start-from);
      const src=ctx.createBufferSource();src.buffer=clip.buffer;const gain=ctx.createGain();gain.gain.value=track.volume;src.connect(gain).connect(ctx.destination);
      const offset=(clip.trimStart||0)+localOffset; const playDur=Math.max(0,dur-localOffset); if(playDur>0){src.start(ctx.currentTime+when,offset,playDur);state.playbackNodes.push(src);}
    }
  }
}

}
