function updatePlayhead(forced=null){
  const t=forced??currentTimelineTime(); const x=state.pxPerSecond*t; els.timelinePlayhead.style.transform=`translateX(${x}px)`;
  const sourceWidth=els.sourceWave.clientWidth; const sourcePct=state.duration?clamp(t/state.duration,0,1):0; els.sourcePlayhead.style.transform=`translateX(${sourceWidth*sourcePct}px)`;
  els.timeReadout.textContent=`${formatTime(t,true)} / ${formatTime(projectEnd(),true)}`;
}

async function beginRecording(){
  if(state.isRecording)return;
  const track=selectedTrack()||createTrack();
  if(!navigator.mediaDevices?.getUserMedia||typeof MediaRecorder==='undefined'){notify('Microphone recording is not supported in this browser.');return;}
  await ensureAudioContext();
  let stream=null;
  try{
    const start=currentTimelineTime();
    const wasPlaying=!!state.isPlaying;
    stream=await navigator.mediaDevices.getUserMedia({audio:{echoCancellation:false,noiseSuppression:false,autoGainControl:false}});
    const preferred=['audio/webm;codecs=opus','audio/ogg;codecs=opus','audio/webm'].find(type=>MediaRecorder.isTypeSupported(type));
    state.recorderStream=stream;state.recordChunks=[];state.discardRecording=false;
    state.recorder=new MediaRecorder(stream,preferred?{mimeType:preferred}:undefined);
    state.recorder.ondataavailable=e=>{if(e.data&&e.data.size)state.recordChunks.push(e.data);};
    state.recorder.onstop=()=>{const discard=state.discardRecording;state.discardRecording=false;if(!discard)finalizeRecording(track);};
    if(els.metronomeToggle.checked){if(state.isPlaying)stopPlayback();await countIn();}
    state.recordStartTime=start;state.isRecording=true;els.recordBtn.textContent='● Recording…';updateButtons();
    state.recorder.start(100);
    if(!state.isPlaying){await startPlayback(start);}
    else if(!wasPlaying){await startPlayback(start);}
    notify('Recording started — source media and microphone are running together.');
  }catch(err){
    console.error(err);
    state.discardRecording=true;state.isRecording=false;
    try{if(state.recorder&&state.recorder.state!=='inactive')state.recorder.stop();}catch{}
    (stream||state.recorderStream)?.getTracks?.().forEach(t=>t.stop());
    state.recorderStream=null;els.recordBtn.textContent='● Record';updateButtons();
    if(err?.name==='NotAllowedError'||err?.name==='SecurityError')notify('Allow microphone access, then press Record again.');
    else notify('Recording could not start. Try Chrome or Edge and make sure the media can play.');
  }
}
async function countIn(){
  const ctx=await ensureAudioContext();
  for(let i=0;i<4;i++){ const osc=ctx.createOscillator(),g=ctx.createGain();osc.frequency.value=i===3?1100:850;g.gain.setValueAtTime(.0001,ctx.currentTime);g.gain.exponentialRampToValueAtTime(.18,ctx.currentTime+.01);g.gain.exponentialRampToValueAtTime(.0001,ctx.currentTime+.12);osc.connect(g).connect(ctx.destination);osc.start();osc.stop(ctx.currentTime+.13);await new Promise(r=>setTimeout(r,500)); }
}
function stopRecording(){
  if(!state.isRecording)return;
  state.isRecording=false;stopPlayback();
  try{if(state.recorder?.state!=='inactive')state.recorder.stop();}catch{}
  state.recorderStream?.getTracks().forEach(t=>t.stop());state.recorderStream=null;
  els.recordBtn.textContent='● Record';updateButtons();
}
async function finalizeRecording(track){
  try{
    if(!state.recordChunks.length){notify('No microphone audio was captured.');return;}
    const blob=new Blob(state.recordChunks,{type:state.recorder?.mimeType||'audio/webm'});const arr=await blob.arrayBuffer();const ctx=await ensureAudioContext();const buffer=await ctx.decodeAudioData(arr.slice(0));
    const clip={id:uid('clip'),name:`Take ${track.clips.length+1}`,buffer,start:state.recordStartTime,trimStart:0,trimEnd:buffer.duration};track.clips.push(clip);state.selectedClipId=clip.id;renderAll();notify(`Recorded ${formatTime(buffer.duration,true)}`);
  }catch(err){console.error(err);notify('Could not decode the recorded take. Try Chrome or Edge.');}
}
async function importAudioToTrack(file,trackId){
  const track=state.tracks.find(t=>t.id===trackId);if(!file||!track)return;const ctx=await ensureAudioContext();
  try{const buffer=await ctx.decodeAudioData((await file.arrayBuffer()).slice(0));const clip={id:uid('clip'),name:file.name,buffer,start:currentTimelineTime(),trimStart:0,trimEnd:buffer.duration};track.clips.push(clip);state.selectedTrackId=track.id;state.selectedClipId=clip.id;renderAll();notify(`Imported ${file.name}`);}catch{notify('That audio format could not be decoded by this browser.');}
}

function splitSelected(){
  const s=selectedClip();if(!s)return;const t=currentTimelineTime(),c=s.clip,d=clipDuration(c),rel=t-c.start;if(rel<=.03||rel>=d-.03){notify('Move the playhead inside the selected clip first.');return;}
  const originalEnd=c.trimEnd;c.trimEnd=(c.trimStart||0)+rel;const right={...c,id:uid('clip'),name:`${c.name} B`,start:t,trimStart:c.trimEnd,trimEnd:originalEnd};s.track.clips.push(right);state.selectedClipId=right.id;renderAll();
}
function trimStart(){const s=selectedClip();if(!s)return;const t=currentTimelineTime(),rel=t-s.clip.start;if(rel<=0||rel>=clipDuration(s.clip)){notify('Move the playhead inside the selected clip.');return;}s.clip.trimStart+=rel;s.clip.start=t;renderAll();}
function trimEnd(){const s=selectedClip();if(!s)return;const t=currentTimelineTime(),rel=t-s.clip.start;if(rel<=0||rel>=clipDuration(s.clip)){notify('Move the playhead inside the selected clip.');return;}s.clip.trimEnd=s.clip.trimStart+rel;renderAll();}
function deleteSelectedClip(){const s=selectedClip();if(!s)return;s.track.clips=s.track.clips.filter(c=>c.id!==s.clip.id);state.selectedClipId=null;renderAll();}
